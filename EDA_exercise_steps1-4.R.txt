# =============================================================================
#### Understanding Q-Q plot #### 
# =============================================================================
# A Quantile-Quantile (Q-Q) plot is a scatter plot used to assess whether a dataset 
# follows a specific theoretical distribution (most commonly a normal distribution) 
# or whether two sample distributions share the same underlying shape.

# Manually create a Q-Q plot based on standard normal distribution (z score)
# Set seed for reproducibility across simulation
set.seed(12)

# Define parameters for standard normal distribution Normal(0, 1)
mu <- 0
sigma <- 1
probabilities <- seq(0.01, 0.99, length.out=1000)
# Theoretical quantiles of expected residuals if they follow a normal distribution
theoretical_z <- qnorm(probabilities, mean = mu, sd = sigma) 
# Empirical quantiles of observed residuals in z scale 
empirical_z <- as.vector(quantile(scale(naive_mod$residuals), probs = probabilities))

# Q-Q plot using standard normal distribution
ggplot() +
  geom_point(aes(x=theoretical_z, y=empirical_z)) +
  geom_line(aes(x=theoretical_z, y=theoretical_z), linetype=1, color="red")

# Manually create a Q-Q plot based on uniform distribution [0,1] (DHARMa's method)
# Emprical quantiles of simulated observed residuals
empirical_u <- sort(naive_sim_res$scaledResiduals)

# Calculate theoretical uniform quantiles (DHARMa's method)
theoretical_u <- (1:length(empirical_u)) / (length(empirical_u) + 1) 

# Q-Q plot from DHARMa using uniform distribution
ggplot() +
  geom_point(aes(x=theoretical_u, y=empirical_u)) +
  geom_line(aes(x=c(0,1), y=c(0,1)), linetype=1, color="red")

#### Understanding residual vs fitted plot ####  
# =============================================================================
# A residuals vs. fitted plot is a scatter plot used to evaluate whether a model 
# satisfies the fundamental assumptions of constant variance (homoscedasticity).

# The residuals vs. fitted plot from the negative binomial model using DHARMa method
# DHARMa's residuals vs. fitted plot uses quantile regressions 
# Quantile regression is a statistical method that estimates specific percentiles,
# such as the 25th percentile, median, or 75th percentile, of a response variable, 
# rather than just its mean
# Quantile regression is a useful tool for exploratory data analysis (EDA) because 
# it allows you to visualize heteroscedasticity without relying on formal statistical 
# tests, which often carry specific assumptions e.g., no outliers, normality of 
# error distribution

# install.packages("DHARMa")
library(DHARMa)
# Count data with group specific dispersion
voles_sim_res <- simulateResiduals(fittedModel = voles_fit, n = 1000)
plotResiduals(voles_sim_res)

# Count data with constant dispersion
voles_sim_res2 <- simulateResiduals(fittedModel = voles_fit2, n = 1000)
plotResiduals(voles_sim_res2)

#extract simulated scaled residuals
voles_nb_resids <- voles_sim_res$scaledResiduals 

#extract the model's fitted values and turn them into scaled ranks 
voles_nb_fitted <- rank(voles_sim_res$fittedPredictedResponse)/(length(voles_sim_res$fittedPredictedResponse)+1) 

# Fit quantile regression at 1st, 2nd and 3rd quanfiles (tau = c(0.25, 0.50, 0.75))
#install.packages("qgam")
library(qgam)

q1_fit <- rq(voles_nb_resids  ~ voles_nb_fitted, tau = 0.25)
q2_fit <- rq(voles_nb_resids  ~ voles_nb_fitted, tau = 0.5)
q3_fit <- rq(voles_nb_resids  ~ voles_nb_fitted, tau = 0.75)

q1_pred <- predict(q1_fit, newdata = data.frame(voles_nb_fitted = voles_nb_fitted), 
                   interval = "confidence", level = 0.95) 
q2_pred <- predict(q2_fit, newdata = data.frame(voles_nb_fitted = voles_nb_fitted), 
                   interval = "confidence", level = 0.95) 
q3_pred <- predict(q3_fit, newdata = data.frame(voles_nb_fitted = voles_nb_fitted), 
                   interval = "confidence", level = 0.95) 


q_pred <- rbind.data.frame(q1_pred, q2_pred, q3_pred) %>% 
  mutate(voles_nb_fitted = rep(voles_nb_fitted, time=3),
         tau=rep(c("Q1", "Q2", "Q3"), each=118))

# Manually create residuals vs fitted plot 
# This plot is similar to but not exactly the DHARMa method, which is too complicate
# for the purpose illustrating residuals vs fitted plot
ggplot() +
  # 1. Plot the raw background residual points
  geom_point(aes(x = voles_nb_fitted, y = voles_nb_resids), color = "gray50", alpha = 0.5) +
  
  # 2. Add the 95% CI Ribbons from your manually built dataframe
  geom_ribbon(data=q_pred, aes(x = voles_nb_fitted, ymin = lower, ymax = higher, fill = tau), 
              alpha = 0.15) +
  
  # 3. Add the precise Quantile Regression Lines 
  geom_line(data = q_pred, 
            aes(x = voles_nb_fitted, y = fit, color = tau), 
            linewidth = 1.2) +
  
  # Custom aesthetic configurations
  scale_color_manual(values = c("Q1" = "blue", "Q2" = "black", "Q3" = "red"),
                     labels = c("25th Percentile", "Median (50th)", "75th Percentile")) +
  scale_fill_manual(values = c("Q1" = "blue", "Q2" = "black", "Q3" = "red"),
                    labels = c("25th Percentile", "Median (50th)", "75th Percentile")) +
  labs(x = "Fitted Values (voles_nb_fitted)", y = "Residuals (voles_nb_resids)")
